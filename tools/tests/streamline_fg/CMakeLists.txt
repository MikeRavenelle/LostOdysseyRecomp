cmake_minimum_required(VERSION 3.28)
project(LoStreamlineFgProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(LO_STREAMLINE_FG_CPU_ONLY "Build input ownership and option contracts without GPU or SDKs" OFF)
enable_testing()
add_executable(LoStreamlineFgProbeOptionsTest probe_options_test.cpp)
add_executable(LoStreamlineFgInputCompletionTest input_completion_test.cpp)
set_target_properties(LoStreamlineFgProbeOptionsTest LoStreamlineFgInputCompletionTest PROPERTIES CXX_SCAN_FOR_MODULES OFF)
add_test(NAME streamline_probe_options COMMAND LoStreamlineFgProbeOptionsTest)
add_test(NAME streamline_input_completion COMMAND LoStreamlineFgInputCompletionTest)
option(LO_STREAMLINE_FG_VULKAN_TEST "Test native synchronization primitives, without Streamline" OFF)
if(LO_STREAMLINE_FG_VULKAN_TEST)
    find_package(Vulkan REQUIRED)
    add_executable(LoStreamlineFgInputVulkanTest input_completion_vulkan_test.cpp)
    target_link_libraries(LoStreamlineFgInputVulkanTest PRIVATE Vulkan::Vulkan)
    set_target_properties(LoStreamlineFgInputVulkanTest PROPERTIES CXX_SCAN_FOR_MODULES OFF)
    add_test(NAME streamline_input_vulkan COMMAND LoStreamlineFgInputVulkanTest)
    set_tests_properties(streamline_input_vulkan PROPERTIES TIMEOUT 60)
endif()
if(LO_STREAMLINE_FG_CPU_ONLY)
    return()
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This is a native Windows x64 Vulkan probe")
endif()
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
include("${ROOT}/cmake/LoDlss.cmake")
if(NOT LO_DLSS_SDK_AVAILABLE OR NOT EXISTS "${LO_DLSS_RUNTIME_FILE}")
    message(FATAL_ERROR "Official local NGX SDK and SR runtime are required: set LO_ENABLE_DLSS=ON, LO_REQUIRE_DLSS=ON, LO_DLSS_SDK_ROOT")
endif()
include("${ROOT}/cmake/LoStreamline.cmake")
add_subdirectory("${ROOT}/thirdparty/plume" plume)
option(LO_STREAMLINE_FG_VALIDATION "Enable the Vulkan validation layer in the standalone probe when installed" ON)
if(LO_STREAMLINE_FG_VALIDATION)
    target_compile_definitions(plume PRIVATE VULKAN_VALIDATION_LAYER_ENABLED=1)
endif()
add_executable(LoStreamlineFgProbe
    streamline_fg_probe.cpp streamline_runtime.cpp probe_vulkan_dispatch.cpp probe_scene.cpp ngx_lifecycle_capture.cpp
    "${ROOT}/LostOdysseyRecomp/gpu/dlss_ngx.cpp")
# Capture raw retirement results even after Controller's bounded call report fills.
set_source_files_properties("${ROOT}/LostOdysseyRecomp/gpu/dlss_ngx.cpp" PROPERTIES COMPILE_DEFINITIONS
    "NVSDK_NGX_VULKAN_ReleaseFeature=LoProbeNgxReleaseFeature;NVSDK_NGX_VULKAN_DestroyParameters=LoProbeNgxDestroyParameters;NVSDK_NGX_VULKAN_Shutdown1=LoProbeNgxShutdown1")
set_target_properties(LoStreamlineFgProbe PROPERTIES CXX_SCAN_FOR_MODULES OFF)
target_compile_options(LoStreamlineFgProbe PRIVATE /utf-8)
target_compile_definitions(LoStreamlineFgProbe PRIVATE LO_GPU_PLUME=1 NOMINMAX)
target_include_directories(LoStreamlineFgProbe PRIVATE "${ROOT}/LostOdysseyRecomp" "${ROOT}/thirdparty/plume")
target_link_libraries(LoStreamlineFgProbe PRIVATE plume d3d12 dxgi wintrust crypt32 psapi bcrypt advapi32)
lo_enable_dlss(LoStreamlineFgProbe)
lo_streamline_probe(LoStreamlineFgProbe)
